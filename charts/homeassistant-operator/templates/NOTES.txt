Quick start — deploy Home Assistant:

  kubectl apply -f - <<EOF
  apiVersion: ha.homeassistant.io/v1
  kind: HomeAssistantConfiguration
  metadata:
    name: home
  spec:
    homeAssistantRef:
      name: home
    configuration: |
      default_config:
  ---
  apiVersion: ha.homeassistant.io/v1
  kind: HomeAssistant
  metadata:
    name: home
  spec:
    version: "2025.latest"  # set to your desired HA version
    bootstrap:
      enabled: true
      credentials:
        secretRef:
          name: ha-admin
  EOF
